# Fix suspend - Gigabyte B550M AORUS ELITE
# Desabilita wake via PCIe Root Port e controlador USB PCIe problemático
#
# GPP0 -> PCI Root Port
# PTXH -> USB xHCI Controller 0000:02:00.0
#
# Solução para:
# - wake imediato após suspend
# - instabilidade ACPI resume
# - xHCI USB resume problemático
#
# ----------------------------------------------------------------------
# VALIDAR ANTES
#
# Verificar se GPP0 e PTXH existem e estão "enabled":
#
cat /proc/acpi/wakeup | grep -E 'GPP0|PTXH'
#
# Caso estejam "enabled", prosseguir com o procedimento abaixo.
# ----------------------------------------------------------------------

sudo nano /etc/systemd/system/disable-gpp0-wake.service

# Conteúdo do arquivo:

[Unit]
Description=Disable problematic PCI wake devices

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStart=/bin/sh -c 'echo GPP0 > /proc/acpi/wakeup; echo PTXH > /proc/acpi/wakeup'

[Install]
WantedBy=multi-user.target

--
sudo systemctl daemon-reload
sudo systemctl enable disable-gpp0-wake.service
sudo systemctl restart disable-gpp0-wake.service

# ----------------------------------------------------------------------
# VALIDAR DEPOIS
#
# GPP0 e PTXH devem aparecer como "disabled":
#
cat /proc/acpi/wakeup | grep -E 'GPP0|PTXH'
# ----------------------------------------------------------------------

# ------------------------------------------------------------
# mDNS / SMB discovery - Dolphin
# Permite resolver hosts .local, ex.: jarvis.local
# ------------------------------------------------------------

echo "[INFO] Configurando mDNS/Avahi para acesso SMB..."

sudo pacman -S --needed --noconfirm \
    avahi \
    nss-mdns \
    openbsd-netcat \
    smbclient \
    kio-extras

# Habilita e inicia o Avahi
sudo systemctl enable --now avahi-daemon.service

# Adiciona suporte a mDNS no NSS
if ! grep -qE '^hosts:.*mdns_minimal' /etc/nsswitch.conf; then
    echo "[INFO] Adicionando mdns_minimal ao /etc/nsswitch.conf..."

    sudo sed -i \
        's/^hosts:.*/hosts: mymachines mdns_minimal [NOTFOUND=return] resolve [!UNAVAIL=return] files myhostname dns/' \
        /etc/nsswitch.conf
else
    echo "[OK] mdns_minimal já configurado no nsswitch.conf."
fi

# Reinicia os serviços envolvidos
sudo systemctl restart avahi-daemon.service

if systemctl is-active --quiet systemd-resolved.service; then
    sudo systemctl restart systemd-resolved.service
fi

echo "[OK] Configuração mDNS/SMB concluída."

############################################################
# Instalação KVM / Libvirt no Arch Linux
############################################################

############################################################
# 1) Confirmar suporte de virtualização
############################################################
grep -Ewo 'vmx|svm' /proc/cpuinfo | sort -u

# Resultado:
# svm = AMD-V
# vmx = Intel VT-x


############################################################
# 2) Instalar os pacotes
############################################################
sudo pacman -S --needed \
  qemu-desktop \
  libvirt \
  virt-install \
  virt-manager \
  virt-viewer \
  edk2-ovmf \
  swtpm \
  libosinfo \
  virtiofsd \
  dnsmasq \
  iptables-nft

# Opcionais para manipulação e inspeção de imagens:
# sudo pacman -S --needed libguestfs guestfs-tools

# Use qemu-full somente quando precisar emular
# outras arquiteturas além de x86/x86_64.


############################################################
# 3) Habilitar os daemons modulares por socket
############################################################
# Este roteiro usa a topologia modular do libvirt.
# Não habilitar libvirtd.socket ao mesmo tempo.

sudo systemctl enable --now \
  virtqemud.socket \
  virtnetworkd.socket \
  virtlogd.socket \
  virtstoraged.socket

systemctl status \
  virtqemud.socket \
  virtnetworkd.socket \
  virtlogd.socket \
  virtstoraged.socket \
  --no-pager


############################################################
# 4) Adicionar o usuário ao grupo libvirt
############################################################
sudo usermod -aG libvirt "$USER"

# Efetuar logout/login ou reiniciar antes dos testes sem sudo.


############################################################
# 5) Configurar a rede NAT padrão
############################################################
if ! sudo virsh net-info default &>/dev/null; then
  sudo virsh net-define /usr/share/libvirt/networks/default.xml
fi

sudo virsh net-start default 2>/dev/null || true
sudo virsh net-autostart default

sudo virsh net-list --all
sudo virsh net-info default


############################################################
# 6) Verificações básicas após novo login
############################################################
id
ls -l /dev/kvm
lsmod | grep -E '^kvm'

virsh -c qemu:///system list --all
virsh -c qemu:///system net-list --all


############################################################
# 7) Opcional: habilitar IOMMU para PCI/GPU passthrough
############################################################
# Antes, habilitar no BIOS/UEFI:
# AMD: SVM e IOMMU
# Intel: Virtualization Technology e VT-d

sudo nano /etc/default/grub

# Acrescentar à linha GRUB_CMDLINE_LINUX_DEFAULT:

# AMD:
# amd_iommu=on iommu=pt

# Intel:
# intel_iommu=on iommu=pt

sudo grub-mkconfig -o /boot/grub/grub.cfg

# Reiniciar o host.


############################################################
# 8) Validar IOMMU após o reboot
############################################################
sudo dmesg | grep -Ei 'AMD-Vi|DMAR|IOMMU'

find /sys/kernel/iommu_groups/ -type l | sort -V


############################################################

############################################################

# - virsh funcionando sem sudo
# - rede NAT default ativa e em autostart
# - firmware OVMF disponível
# - TPM virtual disponível
# - IOMMU disponível para análise de passthrough, se configurado
